% Fig 4(b) (methods, estimation) - leakage-free ensemble-Kalman predict-correct
% loop.
%
% The deployment claim of the paper: the pooled coefficient state is advanced by
% the matched forecast model (predict) and corrected from a sparse set of wall
% pressure sensors through the observation operator (correct). No field ever enters
% the loop; the state is the same 32-vector throughout.
%
% Redrawn 2026-07-30 on Carlos's request. The previous version had three
% crossing connectors, all caused by routing the return path through the middle
% of the diagram:
%   \draw[arr] (zanext) |- (za.west)   ran vertically THROUGH the wall-pressure
%                                      box to reach the state at top left
%   \draw[obsarr] (y) |- (K)           crossed that same return path
% The loop is now a clean rectangular cycle running clockwise, with the return
% leg on the far left where nothing else sits, and the measured pressure
% entering the update from the right so no connector meets another. Descriptive
% text has moved to the caption.
%
% 2026-07-30, second pass: the two slots that distinguish the four estimator
% configurations are annotated in place (propagation F,Q and the observation
% path), because all four are the SAME loop with only those two swapped. The
% per-configuration detail lives in tab:enkf, not here. The stage is named
% "predict" throughout, per the Kalman predict-correct convention; "forecaster"
% is reserved for the \DirectFC operator, which is a different object from the
% co-trained predictor P_phi.
%
% 2026-08-13, third pass (Carlos, comment on S34-C1): the observation side read
% as though the measured pressure carried the innovation. It did not: the
% observation operator maps the FORECAST to a PREDICTED observation, and the
% innovation is that prediction subtracted from the measurement. The previous
% version hung the label "y - Hz^f" on the incoming measurement arrow, with no
% node where the subtraction happens, so the two roles were indistinguishable.
% The difference is now an explicit junction: predicted observation enters from
% above, measurement from the right, innovation leaves to the left into the
% update. Also: the state boxes are annotated "analysis" / "forecast" so the a
% and f superscripts are readable off the figure, and the slot text names the
% spaces rather than abbreviating them.
%
% Standalone-compilable: pdflatex fig_estimation_loop.tex

\documentclass[border=10pt,tikz]{standalone}
\usepackage{tikz}
\usepackage{amsmath,amssymb}
\usetikzlibrary{positioning,arrows.meta,fit,shapes.geometric,calc,backgrounds}

\newcommand{\slot}[1]{{\footnotesize\itshape #1}}

\begin{document}

\begin{tikzpicture}[
    font=\normalsize,
    box/.style={draw, rounded corners=2pt, align=center, inner sep=6pt, thick,
                minimum width=42mm, minimum height=17mm},
    fcst/.style={box, fill=orange!12},
    obsop/.style={box, fill=green!12},
    upd/.style={box, fill=blue!10},
    sensor/.style={box, fill=gray!8, dashed},
    state/.style={draw, rectangle, rounded corners=1pt, fill=white,
                  minimum height=10mm, minimum width=24mm, inner sep=4pt},
    arr/.style={-{Stealth[length=2.6mm]}, thick},
    obsarr/.style={-{Stealth[length=2.6mm]}, thick, green!45!black},
    diff/.style={draw, circle, thick, inner sep=1.5pt, fill=white, minimum size=7mm},
    lbl/.style={font=\footnotesize},
    tagl/.style={font=\footnotesize\itshape, text=black!65}
  ]

  % ===== top edge: predict, left to right =====
  % The state boxes carry "analysis" / "forecast" tags so the a and f
  % superscripts can be read off the figure without the text.
  \node[state] (za) {$\mathbf{z}^{a}_{t}$};
  \node[tagl, above=1mm of za] {analysis at $t$};
  \node[fcst, right=12mm of za] (pred)
    {\textbf{Predict} \\[2pt]
     \slot{$F,\,Q$: matched transformer,} \\
     \slot{linear operator $A$, or direct forecaster}};
  \node[state, right=12mm of pred] (zf) {$\mathbf{z}^{f}_{t+1}$};
  \node[tagl, above=1mm of zf] {forecast at $t+1$};

  \draw[arr] (za) -- (pred);
  \draw[arr] (pred) -- (zf);

  % ===== right edge: the forecast is mapped to a PREDICTED observation =====
  \node[obsop, below=13mm of zf] (H)
    {\textbf{Observation model} $\mathcal{H}$ \\[2pt]
     \slot{$K$ pressure sensors, or} \\ \slot{encoded latent $E_{\mathrm{obs}}$}};
  \draw[arr] (zf) -- (H);

  % ===== the difference junction: this is where the innovation is formed =====
  % Predicted observation from above, measurement from the right. Drawing it as
  % an explicit node is the whole point of the third pass: the innovation is a
  % subtraction, not something the measurement arrives carrying.
  % The measurement enters the junction from BELOW, not from the right: that
  % leaves the whole right-hand side of the vertical arrow free for the
  % predicted-observation label, which is the one piece of text that has to be
  % read to understand what is being subtracted from what.
  \node[diff, below=15mm of H] (sub) {$-$};
  \draw[arr] (H) -- (sub)
    node[midway, right, lbl, align=left]
      {predicted observation \\ $\hat{\mathbf{y}}_{t+1} = \mathcal{H}\mathbf{z}^{f}_{t+1}$};

  \node[sensor, below=15mm of sub] (y)
    {\textbf{Measurement} \\[2pt] \slot{wall pressure $\mathbf{y}_{t+1}$}};
  \draw[obsarr] (y) -- (sub)
    node[midway, right, lbl, black, align=left] {measured at the \\ $K$ sensors};

  % ===== innovation into the update, then the corrected state =====
  \node[upd, left=22mm of sub] (K)
    {\textbf{Update} (correct) \\[2pt] \slot{ensemble Kalman gain}};
  \draw[arr] (sub) -- (K)
    node[midway, above, lbl, align=center]
      {innovation \\ $\boldsymbol{\nu} = \mathbf{y}_{t+1} - \hat{\mathbf{y}}_{t+1}$};

  \node[state] (zanext) at (za |- K) {$\mathbf{z}^{a}_{t+1}$};
  \draw[arr] (K) -- (zanext);
  \node[tagl, below=1mm of zanext] {analysis at $t+1$};

  % ===== left edge: return leg, straight up the outside =====
  \draw[arr] (zanext) -- (za)
    node[midway, left, lbl, align=center] {advance \\ one step};

  \begin{scope}[on background layer]
    \node[draw=orange!35, rounded corners=4pt, dashed, inner sep=4mm,
          fit=(za)(pred)(zf),
          label={[orange!55!black,font=\small]above:Predict: field free, no vorticity enters}] {};
    \node[draw=blue!30, rounded corners=4pt, dashed, inner sep=4mm,
          fit=(zanext)(K)(H)(sub)(y),
          label={[blue!50!black,font=\small]below:Correct: sparse wall pressure only}] {};
  \end{scope}

  % Anchored to the finished bounding box rather than to a node: the measurement
  % box now sits at the bottom right, and any node-relative placement ran the
  % note straight through it.
  \node[anchor=north west, align=left, font=\footnotesize]
    at ([shift={(0,-5mm)}]current bounding box.south west)
    {\textit{Production configuration (two-stage):} direct forecaster with the encoded
     observation \\
     inside the impact window, matched transformer with the pressure sensors outside it.};

\end{tikzpicture}

\end{document}
