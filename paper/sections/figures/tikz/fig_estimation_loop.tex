% Fig 4(b) (methods, estimation) - leakage-free ensemble-Kalman predict-correct
% loop.
%
% The deployment claim of the paper: the pooled coefficient state is advanced by
% the matched forecast model (predict) and corrected from a sparse set of wall
% pressure taps through the observation operator (correct). No field ever enters
% the loop; the state is the same 32-vector throughout.
%
% Redrawn 2026-07-30 on Carlos's request. The previous version had three
% crossing connectors, all caused by routing the return path through the middle
% of the diagram:
%   \draw[arr] (zanext) |- (za.west)   ran vertically THROUGH the wall-pressure
%                                      box to reach the state at top left
%   \draw[obsarr] (y) |- (K)           crossed that same return path
% The loop is now a clean rectangular cycle running clockwise, with the return
% leg on the far left where nothing else sits, and the measured pressure
% entering the update from the right so no connector meets another. Descriptive
% text has moved to the caption.
%
% 2026-07-30, second pass: the two slots that distinguish the four estimator
% configurations are annotated in place (propagation F,Q and the observation
% path), because all four are the SAME loop with only those two swapped. The
% per-configuration detail lives in tab:enkf, not here. The stage is named
% "predict" throughout, per the Kalman predict-correct convention; "forecaster"
% is reserved for the \DirectFC operator, which is a different object from the
% co-trained predictor P_phi.
%
% Standalone-compilable: pdflatex fig_estimation_loop.tex

\documentclass[border=10pt,tikz]{standalone}
\usepackage{tikz}
\usepackage{amsmath,amssymb}
\usetikzlibrary{positioning,arrows.meta,fit,shapes.geometric,calc,backgrounds}

\newcommand{\slot}[1]{{\footnotesize\itshape #1}}

\begin{document}

\begin{tikzpicture}[
    font=\normalsize,
    box/.style={draw, rounded corners=2pt, align=center, inner sep=6pt, thick,
                minimum width=42mm, minimum height=17mm},
    fcst/.style={box, fill=orange!12},
    obsop/.style={box, fill=green!12},
    upd/.style={box, fill=blue!10},
    sensor/.style={box, fill=gray!8, dashed},
    state/.style={draw, rectangle, rounded corners=1pt, fill=white,
                  minimum height=10mm, minimum width=24mm, inner sep=4pt},
    arr/.style={-{Stealth[length=2.6mm]}, thick},
    obsarr/.style={-{Stealth[length=2.6mm]}, thick, green!45!black},
    lbl/.style={font=\footnotesize}
  ]

  % ===== top edge: predict, left to right =====
  \node[state] (za) {$\mathbf{z}^{a}_{t}$};
  \node[fcst, right=12mm of za] (pred)
    {\textbf{Predict} \\[2pt]
     \slot{$F,\,Q$: transformer,} \\ \slot{linear $A$, or direct forecaster}};
  \node[state, right=12mm of pred] (zf) {$\mathbf{z}^{f}_{t+1}$};

  \draw[arr] (za) -- (pred);
  \draw[arr] (pred) -- (zf);

  % ===== right edge: observation operator =====
  \node[obsop, below=14mm of zf] (H)
    {\textbf{Observation} $\mathcal{H}$ \\[2pt]
     \slot{$K$ taps (tap space), or} \\ \slot{encoded $E_{\mathrm{obs}}$ (state space)}};
  \draw[arr] (zf) -- (H) node[midway, right, lbl] {$\mathcal{H}\mathbf{z}^{f}$};

  % ===== bottom right: update =====
  \node[upd, below=14mm of H] (K) {\textbf{Update} (correct)};
  \draw[arr] (H) -- (K);

  % measured pressure enters the update from the RIGHT: nothing else is there,
  % so this connector crosses nothing
  \node[sensor, right=12mm of K] (y) {wall pressure $\mathbf{y}_{t+1}$};
  \draw[obsarr] (y) -- (K)
    node[midway, above, lbl, black] {$\mathbf{y}-\mathcal{H}\mathbf{z}^{f}$};

  % ===== bottom edge: corrected state, directly below z^a_t =====
  \node[state] (zanext) at (za |- K) {$\mathbf{z}^{a}_{t+1}$};
  \draw[arr] (K) -- (zanext);

  % ===== left edge: return leg, straight up the outside =====
  \draw[arr] (zanext) -- (za)
    node[midway, left, lbl, align=center] {advance \\ one step};

  \node[below=13mm of zanext.south, anchor=north west, align=left,
        font=\footnotesize, xshift=-12mm] (note)
    {\textit{Production (two-stage):} direct forecaster $+$ encoded $E_{\mathrm{obs}}$ \\
     inside the impact window, matched transformer $+$ tap space outside.};

  \begin{scope}[on background layer]
    \node[draw=orange!35, rounded corners=4pt, dashed, inner sep=4mm,
          fit=(za)(pred)(zf),
          label={[orange!55!black,font=\small]above:Predict: field free, no vorticity enters}] {};
    \node[draw=blue!30, rounded corners=4pt, dashed, inner sep=4mm,
          fit=(zanext)(K)(H)(y),
          label={[blue!50!black,font=\small]below:Correct: sparse wall pressure}] {};
  \end{scope}

\end{tikzpicture}

\end{document}
