% Fig 4(a) - the trained framework: encoder, predictor, supervision heads,
% visualisation decoder.
%
% Redrawn 2026-07-30 on Carlos's request. The previous version (v6) was a tall
% narrow column: at 0.46\linewidth its type rendered very small, and the boxes
% carried configuration detail plus two italic side notes. It is now a wide
% left-to-right flow, so it sets at full \linewidth with much larger type,
% every architectural detail has moved to the caption, and each module states
% the stages it is ACTIVE in.
%
% Stage attribution (verified against the methods text; do not change casually):
%   encoder E_theta   train + analysis  not used at deployment: the estimator is
%                                       pressure-only and no field enters it
%   heads L, W, N     train only        never part of the latent-prediction loss
%   predictor P_phi   train + deploy    the estimator's forecast step refits a
%                                       predictor of this same class
%   decoder D_psi     analysis only     trained after the encoder is frozen
% The deployment-side observation map E_obs (tap history -> state, eq. eq:eobs)
% is a separate ridge map and belongs to panel (b), NOT here.
%
% Standalone-compilable: pdflatex fig1_jepa_architecture.tex

\documentclass[border=10pt,tikz]{standalone}
\usepackage{tikz}
\usepackage{amsmath,amssymb}
\usetikzlibrary{positioning,arrows.meta,fit,calc,backgrounds}

\newcommand{\stg}[1]{{\scriptsize\itshape #1}}

\begin{document}

\begin{tikzpicture}[
    font=\normalsize,
    stg/.style={font=\scriptsize\itshape},
    mod/.style={draw, rounded corners=2pt, align=center, inner sep=6pt, thick,
                minimum width=34mm, minimum height=14mm},
    enc/.style={mod, fill=blue!10},
    prd/.style={mod, fill=orange!12},
    dcd/.style={mod, fill=gray!8, dashed},
    head/.style={draw, rounded corners=2pt, align=center, inner sep=4pt, thick,
                 fill=green!12, minimum width=31mm, minimum height=9mm},
    io/.style={draw, rounded corners=1pt, fill=white, align=center,
               inner sep=4pt, minimum height=10mm, minimum width=24mm},
    arr/.style={-{Stealth[length=2.6mm]}, thick},
    darr/.style={-{Stealth[length=2.6mm]}, thick, dashed, gray!60!black}
  ]

  % ---- main left-to-right flow ----
  \node[io] (w) {$\boldsymbol{\omega}_z(t)$};
  \node[enc, right=9mm of w] (E)
    {\textbf{Encoder} $E_{\theta}$ \\[1pt] \stg{train $+$ analysis}};
  \node[io, right=9mm of E] (z) {$\mathbf{z}_t \in \mathbb{R}^{d}$};
  \node[prd, right=9mm of z] (P)
    {\textbf{Predictor} $P_{\phi}$ \\[1pt] \stg{train $+$ deploy}};
  \node[io, right=9mm of P] (zh) {$\widehat{\mathbf{z}}_{t+\Delta t}$};

  \draw[arr] (w) -- (E);
  \draw[arr] (E) -- (z);
  \draw[arr] (z) -- (P);
  \draw[arr] (P) -- (zh);

  % ---- supervision heads, above the spine (training only) ----
  \node[head] (hW) at ($(z.north)+(0,20mm)$) {$W$ : wake};
  \node[head, left=6mm of hW]  (hL) {$L$ : lift};
  \node[head, right=6mm of hW] (hN) {$N$ : near-body};
  % one group label for all three heads, so "train only" cannot be misread as
  % qualifying the middle head alone
  \begin{scope}[on background layer]
    \node[draw=green!45!black, rounded corners=3pt, dashed, inner sep=2.5mm,
          fit=(hL)(hW)(hN),
          label={[font=\scriptsize\itshape, green!35!black]above:train only}] {};
  \end{scope}

  \draw[arr] (z.north) -- (hW.south);
  \draw[arr] (z.north) |- ($(hL.south)+(0,-5mm)$) -- (hL.south);
  \draw[arr] (z.north) |- ($(hN.south)+(0,-5mm)$) -- (hN.south);

  % ---- visualisation decoder, below the spine (analysis only) ----
  \node[dcd] (D) at ($(z.south)+(0,-20mm)$)
    {\textbf{Decoder} $D_{\psi}$ \\[1pt] \stg{analysis only}};
  \node[io, right=9mm of D] (wh) {$\widehat{\boldsymbol{\omega}}_z$};

  \draw[darr] (z.south) -- (D.north);
  \draw[darr] (D) -- (wh);

\end{tikzpicture}

\end{document}
